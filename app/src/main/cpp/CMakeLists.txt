cmake_minimum_required(VERSION 3.22)
project(nama2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
include(FetchContent)
# Override FETCHCONTENT_SOURCE_DIR_OBOE / FETCHCONTENT_SOURCE_DIR_NAMCORE for offline builds.
FetchContent_Declare(oboe GIT_REPOSITORY https://github.com/google/oboe.git GIT_TAG 1.9.3 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(oboe)
FetchContent_Declare(namcore GIT_REPOSITORY https://github.com/sdatkinson/NeuralAmpModelerCore.git GIT_TAG v0.5.2 GIT_SHALLOW TRUE GIT_SUBMODULES Dependencies/eigen)
FetchContent_GetProperties(namcore)
if(NOT namcore_POPULATED)
    FetchContent_Populate(namcore)
endif()
# Upstream root builds desktop tools; compile only the inference runtime on Android.
file(GLOB NAM_SOURCES CONFIGURE_DEPENDS "${namcore_SOURCE_DIR}/NAM/*.cpp" "${namcore_SOURCE_DIR}/NAM/*.c" "${namcore_SOURCE_DIR}/NAM/*/*.cpp")
add_library(nam_runtime STATIC ${NAM_SOURCES})
target_include_directories(nam_runtime PUBLIC "${namcore_SOURCE_DIR}" "${namcore_SOURCE_DIR}/Dependencies/eigen" "${namcore_SOURCE_DIR}/Dependencies/nlohmann")
target_compile_definitions(nam_runtime PUBLIC NAM_SAMPLE_FLOAT NAM_ENABLE_A2_FAST)
set_target_properties(nam_runtime PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(nama2 SHARED Engine.cpp)
target_link_libraries(nama2 PRIVATE oboe nam_runtime android log)
target_compile_options(nama2 PRIVATE -Wall -Wextra)
